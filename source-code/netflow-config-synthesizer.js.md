Sends an RFC 7011 IPFIX (v10) / NetFlow v9 packet with Template FlowSet (ID 256\) and Data FlowSet.  
"""

import socket  
import struct  
import time  
import sys

def build\_ipfix\_packet():  
    \# 1\. IPFIX Header: Version(2), Length(2), ExportTime(4), Seq(4), ObsDomain(4) \= 16 bytes  
    version \= \${protocolVersion}  
    export\_time \= int(time.time())  
    seq\_number \= 1001  
    obs\_domain \= \${cfg.observationDomainId || 1}

    \# Template FlowSet: SetID=2, Length=52, TemplateID=256, FieldCount=12  
    \# Fields: (8,4), (12,4), (7,2), (11,2), (4,1), (6,2), (10,4), (14,4), (1,8), (2,8), (152,8), (153,8)  
    template\_fields \= \[  
        (8, 4),   \# sourceIPv4Address  
        (12, 4),  \# destinationIPv4Address  
        (7, 2),   \# sourceTransportPort  
        (11, 2),  \# destinationTransportPort  
        (4, 1),   \# protocolIdentifier  
        (6, 2),   \# tcpControlBits  
        (10, 4),  \# ingressInterface  
        (14, 4),  \# egressInterface  
        (1, 8),   \# octetDeltaCount  
        (2, 8),   \# packetDeltaCount  
        (152, 8), \# flowStartMilliseconds  
        (153, 8\)  \# flowEndMilliseconds  
    \]  
      
    template\_record \= struct.pack("\!HH", 256, len(template\_fields))  
    for f\_id, f\_len in template\_fields:  
        template\_record \+= struct.pack("\!HH", f\_id, f\_len)  
      
    template\_set\_len \= 4 \+ len(template\_record)  
    template\_set \= struct.pack("\!HH", 2, template\_set\_len) \+ template\_record

    \# Data FlowSet: SetID=256, Length=4 \+ RecordLength  
    src\_ip \= socket.inet\_aton("10.200.4.1")  
    dst\_ip \= socket.inet\_aton("198.51.100.44")  
    src\_port \= 53248  
    dst\_port \= 443  
    proto \= 6 \# TCP  
    flags \= 0x0012 \# SYN+ACK  
    in\_if \= 2  
    out\_if \= 4  
    octets \= 4999942144  
    pkts \= 3448850  
    start\_ms \= int(time.time() \* 1000\) \- 30000  
    end\_ms \= int(time.time() \* 1000\)

    data\_record \= (  
        src\_ip \+ dst\_ip \+  
        struct.pack("\!HHBBIIQQQQ", src\_port, dst\_port, proto, 0, in\_if, out\_if, octets, pkts, start\_ms, end\_ms)  
    )  
    \# Fix alignment  
    data\_record \= (  
        src\_ip \+ dst\_ip \+  
        struct.pack("\!HH", src\_port, dst\_port) \+  
        struct.pack("\!B", proto) \+  
        struct.pack("\!H", flags) \+  
        struct.pack("\!II", in\_if, out\_if) \+  
        struct.pack("\!QQQQ", octets, pkts, start\_ms, end\_ms)  
    )  
    pad \= (4 \- (len(data\_record) % 4)) % 4

    finally:
